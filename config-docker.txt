#add ihomis-reports.docker.config in laragon nginx -> sites-enabled
#ihomis-reports.docker.config
server {
    listen 443 ssl;
    server_name ihomis-reports.docker;

    ssl_certificate      C:/laragon/etc/ssl/laragon.crt;
	ssl_certificate_key  C:/laragon/etc/ssl/laragon.key;

    location / {
        proxy_pass https://127.0.0.1:2443;
        proxy_ssl_verify off;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}

#Restart the container
docker restart ihomis-reports







#Build image
docker build -t ihomis-reports-app:v1.1 .
docker build -t ihomis-reports-app:nlv1.1 .
